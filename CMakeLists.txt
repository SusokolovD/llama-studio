# Some relocated Qt 5 static kits retain absolute OpenSSL import-library paths
# in their .prl files. Replace those dependencies in memory; never edit the kit.
function(studio_resolve_static_openssl)
  if(NOT MINGW OR NOT LLAMA_STUDIO_STATIC_QT OR NOT LLAMA_STUDIO_STATIC_OPENSSL_DIR)
    return()
  endif()
  foreach(_name ssl crypto)
    if(NOT EXISTS "${LLAMA_STUDIO_STATIC_OPENSSL_DIR}/lib${_name}.a")
      message(FATAL_ERROR "Missing static OpenSSL archive: ${LLAMA_STUDIO_STATIC_OPENSSL_DIR}/lib${_name}.a")
    endif()
  endforeach()
  get_property(_imported_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
  get_cmake_property(_cache_variables CACHE_VARIABLES)
  foreach(_target IN LISTS _imported_targets)
    get_target_property(_links ${_target} INTERFACE_LINK_LIBRARIES)
    if(NOT _links)
      continue()
    endif()
    foreach(_variable IN LISTS _cache_variables)
      if(_variable MATCHES "^_Qt5.*_(RELEASE|DEBUG)_lib(ssl|crypto)_PATH$")
        set(_name "${CMAKE_MATCH_2}")
        set(_replacement "${LLAMA_STUDIO_STATIC_OPENSSL_DIR}/lib${_name}.a")
        if(_name STREQUAL "ssl")
          # libssl references libcrypto, so rescan crypto after ssl on GNU ld.
          list(APPEND _replacement "${LLAMA_STUDIO_STATIC_OPENSSL_DIR}/libcrypto.a")
        endif()
        if(WIN32)
          list(APPEND _replacement ws2_32 crypt32 advapi32 user32 gdi32)
        endif()
        string(REPLACE "${${_variable}}" "${_replacement}" _links "${_links}")
      endif()
    endforeach()
    set_property(TARGET ${_target} PROPERTY INTERFACE_LINK_LIBRARIES "${_links}")
  endforeach()
endfunction()

function(studio_import_plugins target)
  if(LLAMA_STUDIO_STATIC_QT AND WIN32)
    if(NOT TARGET Qt${QT_VERSION_MAJOR}::QWindowsIntegrationPlugin)
      message(FATAL_ERROR "Static Qt is missing QWindowsIntegrationPlugin")
    endif()
    if(QT_VERSION_MAJOR EQUAL 5)
      qt5_import_plugins(${target} INCLUDE_BY_TYPE platforms Qt5::QWindowsIntegrationPlugin Qt5::QOffscreenIntegrationPlugin)
    else()
      qt6_import_plugins(${target} INCLUDE_BY_TYPE platforms Qt6::QWindowsIntegrationPlugin Qt6::QOffscreenIntegrationPlugin)
      if(TARGET Qt6::QSchannelBackendPlugin)
        qt6_import_plugins(${target} INCLUDE Qt6::QSchannelBackendPlugin)
      endif()
    endif()
  endif()
endfunction()
